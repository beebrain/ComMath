% 09 — XOR gate with truth table and symbolic definition
\begin{tikzpicture}[line width=1pt,>={Stealth[length=2.4mm]},font=\itshape]
  \ctikzset{bipoles/length=0.9cm}
  \node[xor gate US,draw,fill=lpurple,draw=vpurple,line width=1.4pt,scale=2.3] (G) at (3.8,0) {};
  \node[font=\large\upshape] at ([xshift=2pt]G.center) {$\oplus$};
  % logic inputs p, q
  \draw[navy,line width=1.2pt] (0.5,0.65) node[left,black,font=\large\itshape]{$p$} -- (G.input 1);
  \draw[navy,line width=1.2pt] (0.5,-0.65) node[left,black,font=\large\itshape]{$q$} -- (G.input 2);
  \draw[navy,line width=1.2pt] (G.output) -- ++(1,0) to[lamp,l_=\upshape output] ++(2.2,0);
  % truth table to the right
  \matrix[ttmatrix,right=6mm of G.east,anchor=west,yshift=-2mm] {
    |[tthead]| $p$ \& |[tthead]| $q$ \& |[tthead]| $p\oplus q$ \\
    |[TvalR]| T \& |[TvalR]| T \& |[FvalR]| F \\
    |[TvalG]| T \& |[FvalG]| F \& |[TvalG]| T \\
    |[FvalG]| F \& |[TvalG]| T \& |[TvalG]| T \\
    |[FvalR]| F \& |[FvalR]| F \& |[FvalR]| F \\
  };
  \node[formula,below=11mm of G.south,anchor=north,xshift=14mm,font=\itshape]
        {$p \oplus q \;\equiv\; (p \lor q) \land \lnot(p \land q)$};
\end{tikzpicture}
