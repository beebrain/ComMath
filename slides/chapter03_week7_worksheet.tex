%% ---------- กิจกรรม 1 ----------
\noindent\textbf{กิจกรรมที่ 1 : พื้นฐานคู่อันดับ ผลคูณคาร์ทีเซียน และการหาจำนวนความสัมพันธ์} \hfill \textit{(6 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item กำหนด $A = \{1, 2, 3\}$ และ $B = \{x, y\}$ จงหา:
        \quad (ก) $|A \times B| =$ \makebox[2cm]{\dotfill}
        \quad (ข) จำนวนความสัมพันธ์ทั้งหมดจาก $A$ ไป $B = 2^{|A \times B|} =$ \makebox[2.5cm]{\dotfill}
        \quad (ค) จำนวนความสัมพันธ์ทวิภาคทั้งหมดบนเซต $A = 2^{|A|^2} =$ \makebox[3cm]{\dotfill}
  \item กำหนด $S = \{1, 2, 3, 4\}$ และความสัมพันธ์ $R = \{(a, b) \in S \times S \mid a < b\}$\\
        จงแจกแจงสมาชิกของ $R = $ \makebox[13cm]{\dotfill}
  \item จงหาโดเมนและเรนจ์ของ $R$ ในข้อ 2):
        \quad $\operatorname{Dom}(R) = $ \makebox[3.5cm]{\dotfill}
        \quad $\operatorname{Ran}(R) = $ \makebox[3.5cm]{\dotfill}
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 2 ----------
\noindent\textbf{กิจกรรมที่ 2 : ความสัมพันธ์ผกผันและการดำเนินการระหว่างความสัมพันธ์} \hfill \textit{(6 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item กำหนด $R_1 = \{(1, a), (2, b), (3, a)\}$ จาก $\{1, 2, 3\}$ ไป $\{a, b\}$\\
        จงหาความสัมพันธ์ผกผัน $R_1^{-1} = $ \makebox[12cm]{\dotfill}
  \item กำหนด $R_2 = \{(1, b), (2, b), (3, a)\}$ จงหา:
        \begin{enumerate}[label=(\alph*),itemsep=1pt]
          \item $R_1 \cup R_2 =$ \makebox[12cm]{\dotfill}
          \item $R_1 \cap R_2 =$ \makebox[12cm]{\dotfill}
          \item $R_1 - R_2 =$ \makebox[12cm]{\dotfill}
        \end{enumerate}
\end{enumerate}
\vspace{0.2em}

\noindent\textbf{กิจกรรมที่ 3 : การตรวจสอบสมบัติของความสัมพันธ์ทวิภาค (Reflexive, Symmetric, Transitive)} \hfill \textit{(8 นาที)}\\

$R_A = \{(1,1), (2,2), (3,3), (1,2), (2,1)\}$ & & & & \\ \hline
$R_B = \{(1,2), (2,3), (1,3)\}$ & & & & \\ \hline

%% ---------- กิจกรรม 4 ----------
\noindent\textbf{กิจกรรมที่ 4 : คลาสสมมูลและการแบ่งกั้นเซต (Equivalence Class \& Partition)} \hfill \textit{(7 นาที)}\\
กำหนด $A = \{0, 1, 2, 3, 4, 5, 6, 7\}$ และความสัมพันธ์สมมูล $R = \{(a, b) \in A \times A \mid a \equiv b \pmod 3\}$
\begin{enumerate}[label=\arabic*)]
  \item จงแจกแจงคลาสสมมูลทั้งหมดของ $R$ บนเซต $A$:
        \begin{enumerate}[label=(\alph*),itemsep=2pt]
          \item $[0] = \{x \in A \mid x \equiv 0 \pmod 3\} =$ \makebox[8cm]{\dotfill}
          \item $[1] = \{x \in A \mid x \equiv 1 \pmod 3\} =$ \makebox[8cm]{\dotfill}
          \item $[2] = \{x \in A \mid x \equiv 2 \pmod 3\} =$ \makebox[8cm]{\dotfill}
        \end{enumerate}
  \item ตรวจสอบว่า $[0] \cup [1] \cup [2] = A$ และแต่ละคลาสไม่มีสมาชิกร่วมกัน ($[i] \cap [j] = \emptyset$) หรือไม่ $\Rightarrow$ \makebox[3cm]{\dotfill}
  \item สรุป: คลาสสมมูลเหล่านี้สร้าง \textbf{การแบ่งกั้น (Partition)} ของเซต $A$ ออกเป็น \makebox[1.5cm]{\dotfill} กลุ่ม
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 5 ----------
\noindent\textbf{กิจกรรมที่ 5 : การประกอบความสัมพันธ์และเมทริกซ์ 0-1 (Composite Relations \& 0-1 Matrix)} \hfill \textit{(8 นาที)}\\
กำหนด $A = \{1, 2, 3\}, B = \{a, b, c\}, C = \{x, y\}$ โดย $R = \{(1, a), (1, b), (2, c), (3, a)\}$ และ $S = \{(a, x), (b, y), (c, x)\}$
\begin{enumerate}[label=\arabic*)]
  \item จงหาความสัมพันธ์ประกอบ $S \circ R$ (จาก $A$ ไป $C$ โดยผ่าน $B$):\\[0.15em]
        $S \circ R =$ \makebox[13cm]{\dotfill}
  \item เขียนเมทริกซ์บูลีน $M_R$ (ขนาด $3 \times 3$ ของแถว $1,2,3$ และคอลัมน์ $a,b,c$) และ $M_S$ (แถว $a,b,c$ และคอลัมน์ $x,y$):
        \[
        M_R = \begin{bmatrix}
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} \\
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} \\
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill}
        \end{bmatrix}, \quad
        M_S = \begin{bmatrix}
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} \\
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} \\
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill}
        \end{bmatrix}
        \quad \implies M_{S \circ R} = M_R \odot M_S = \begin{bmatrix}
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} \\
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill} \\
        \makebox[0.5cm]{\dotfill} & \makebox[0.5cm]{\dotfill}
        \end{bmatrix}
        \]
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 6 ----------
\noindent\textbf{กิจกรรมที่ 6 : การปิดสมบัติของความสัมพันธ์ (Closures of Relations)} \hfill \textit{(7 นาที)}\\

\end{enumerate}
