%% ---------- กิจกรรม 1 ----------
\noindent\textbf{กิจกรรมที่ 1 : เพาเวอร์เซตและความสัมพันธ์เชิงโครงสร้าง} \hfill \textit{(7 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item กำหนดเซต $A = \{1, \{2, 3\}\}$ จงแจกแจงสมาชิกทุกตัวของ $\mathcal{P}(A)$ ให้ครบถ้วน\\[0.2em]
        \textcolor{gray!60}{\hrulefill}
  \item กำหนด $B = \{a, b, c, d\}$ จงหา:
        \begin{enumerate}[label=(\alph*),itemsep=2pt]
          \item $|\mathcal{P}(B)| =$ \makebox[3cm]{\dotfill}
          \item จำนวนเซตย่อยของ $B$ ที่มีขนาด 2 สมาชิกพอดี $=$ \makebox[2.5cm]{\dotfill}
          \item จำนวนเซตย่อยของ $B$ ที่ \textbf{มี $a$ เป็นสมาชิก} $=$ \makebox[2.5cm]{\dotfill}
        \end{enumerate}
  \item ใส่เครื่องหมาย \textbf{T} (จริง) หรือ \textbf{F} (เท็จ) หน้าข้อความต่อไปนี้:
        \quad [ \ ] $\emptyset \in \mathcal{P}(\mathcal{P}(\emptyset))$ \quad
        \quad [ \ ] $\{\emptyset\} \in \mathcal{P}(\mathcal{P}(\emptyset))$ \quad
        \quad [ \ ] $|\mathcal{P}(\mathcal{P}(\{1\}))| = 4$
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 2 ----------
\noindent\textbf{กิจกรรมที่ 2 : ผลคูณคาร์ทีเซียนและการประยุกต์คู่อันดับ} \hfill \textit{(7 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item ถ้า $(3x + y, \ 8) = (11, \ 2x - y)$ จงหาค่าของ $x$ และ $y$\\[0.2em]
        \textcolor{gray!60}{\hrulefill}
  \item กำหนด $S = \{0, 1, 2\}$ จงเขียนเซต $R = \{(a, b) \in S \times S \mid a + b = 2\}$ แบบแจกแจงสมาชิก\\[0.2em]
        \textcolor{gray!60}{\hrulefill}
  \item ถ้า $|A| = 3$ และ $|B| = 4$ จงหาค่าของ $|(A \times B) \cup (B \times A)|$ เมื่อ $A \cap B = \emptyset$\\[0.2em]
        \textcolor{gray!60}{\hrulefill}
\end{enumerate}
\vspace{0.2em}

        \textcolor{gray!60}{\hrulefill}

        \textcolor{gray!60}{\hrulefill}\\[0.6em]
        \textcolor{gray!60}{\hrulefill}
\end{enumerate}
