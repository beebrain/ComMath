\noindent\textbf{กิจกรรมที่ 1 : ลำดับเลขคณิตและเรขาคณิตในชีวิตประจำวัน (Arithmetic \& Geometric Sequences)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item \textbf{การออมเงินแบบฝากเพิ่มรายเดือน (ลำดับเลขคณิต):} เปิดบัญชีเริ่มต้น 1,000 บาท ($a_1 = 1,000$) และฝากเพิ่มเดือนละ 500 บาทเท่ากันทุกเดือน ($1000, 1500, 2000, \dots$)
        \begin{enumerate}[label=(\alph*),itemsep=1pt]
          \item ผลต่างร่วม $d =$ \makebox[2cm]{\dotfill} บาท
          \item เขียนสูตรพจน์ทั่วไป $a_n = a_1 + (n-1)d =$ \makebox[4cm]{\dotfill}
          \item เมื่อครบ 1 ปี (สิ้นเดือนที่ 12) จะมียอดเงินในบัญชี $a_{12} =$ \makebox[3cm]{\dotfill} บาท
        \end{enumerate}
  \item กำหนดลำดับเลขคณิต $3, 7, 11, 15, \dots$
        \quad (ก) $d =$ \makebox[1.5cm]{\dotfill}
        \quad (ข) $a_n =$ \makebox[3cm]{\dotfill}
        \quad (ค) $a_{50} =$ \makebox[2cm]{\dotfill}
  \item \textbf{การเติบโตของเงินลงทุนแบบทบเท่าตัว (ลำดับเรขาคณิต):} เงินต้น 10,000 บาท ($a_1 = 10,000$) มูลค่าเพิ่มขึ้นเป็น 2 เท่าทุกปี ($10000, 20000, 40000, \dots$)
        \begin{enumerate}[label=(\alph*),itemsep=1pt]
          \item อัตราส่วนร่วม $r =$ \makebox[2cm]{\dotfill}
          \item เขียนสูตรพจน์ทั่วไป $a_n = a_1 \cdot r^{n-1} =$ \makebox[4cm]{\dotfill}
          \item สิ้นปีที่ 5 เงินลงทุนจะมีมูลค่า $a_5 =$ \makebox[3cm]{\dotfill} บาท
        \end{enumerate}
  \item กำหนดลำดับเรขาคณิต $2, 6, 18, 54, \dots$
        \quad (ก) $r =$ \makebox[1.5cm]{\dotfill}
        \quad (ข) $a_n =$ \makebox[3cm]{\dotfill}
        \quad (ค) $a_8 =$ \makebox[2cm]{\dotfill}
\end{enumerate}

%% ---------- กิจกรรม 2 ----------
\noindent\textbf{กิจกรรมที่ 2 : อนุกรมและผลรวมสะสม (Series \& Summations)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item \textbf{สัญกรณ์ซิกมา ($\sum$):} จงเขียนกระจายพจน์และหาผลลัพธ์ของ $\sum_{i=1}^4 (3i + 2)$\\[0.15em]
        กระจายพจน์: \makebox[7cm]{\dotfill} $=$ \makebox[2.5cm]{\dotfill}
  \item \textbf{ชาเลนจ์ออมเงิน 100 วัน (เทคนิคการจับคู่ของเกาส์):}\\
        วันที่ 1 ออม 1 บาท จนถึงวันที่ 100 ออม 100 บาท ($S = 1 + 2 + 3 + \dots + 100$)\\
        จงแสดงวิธีคิดด้วยสูตร $S_n = \frac{n}{2}(a_1 + a_n)$:\\[0.15em]
        $S_{100} = \frac{100}{2}(\makebox[1.2cm]{\dotfill} + \makebox[1.2cm]{\dotfill}) = 50 \times \makebox[1.5cm]{\dotfill} =$ \makebox[3cm]{\dotfill} บาท
  \item \textbf{แผนการออมเงิน 20 วัน:} ออมวันแรก 5 บาท หยอดเพิ่มวันละ 3 บาท ($5 + 8 + 11 + 14 + \dots$)\\
        จงหาผลรวมเงินออม 20 วันแรก ($S_{20}$) ด้วยสูตร $S_n = \frac{n}{2}[2a_1 + (n - 1)d]$:\\[0.15em]
        $S_{20} = \frac{20}{2}[2(5) + (20-1)(3)] = 10[10 + \makebox[1.5cm]{\dotfill}] = 10 \times \makebox[1.5cm]{\dotfill} =$ \makebox[2.5cm]{\dotfill} บาท
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 3 ----------
\noindent\textbf{กิจกรรมที่ 3 : อนุกรมเรขาคณิตจำกัดและอนันต์ (Finite \& Infinite Geometric Series)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item \textbf{ผลรวมอนุกรมเรขาคณิตจำกัด:} แผนการออมเงินแบบทวีคูณ $3 + 6 + 12 + 24 + \dots$ จงหาผลรวม 6 วันแรก ($S_6$)\\[0.15em]
        วิธีทำ: $a_1 = 3, \ r = $ \makebox[1.5cm]{\dotfill} , \quad สูตร $S_6 = \frac{a_1(r^6 - 1)}{r - 1} =$ \makebox[4cm]{\dotfill} $= $ \makebox[2.5cm]{\dotfill} บาท
  \item \textbf{การพิจารณาภาวะลู่เข้าและการลู่ออก (Convergence vs Divergence):} จงระบุอัตราส่วนร่วม $r$ และทำเครื่องหมาย $\checkmark$
        \begin{enumerate}[label=(\alph*),itemsep=2pt]
          \item การแบ่งเค้กทีละครึ่ง: $\frac{1}{2} + \frac{1}{4} + \frac{1}{8} + \dots \implies r = $ \makebox[1.5cm]{\dotfill} \quad [ \ ] ลู่เข้า (Convergent) \quad [ \ ] ลู่ออก (Divergent)
          \item การออมเงินทวีคูณไม่จำกัด: $1 + 2 + 4 + 8 + \dots \implies r = $ \makebox[1.5cm]{\dotfill} \quad [ \ ] ลู่เข้า (Convergent) \quad [ \ ] ลู่ออก (Divergent)
          \item อนุกรมแกว่งกวัด: $5 - 5 + 5 - 5 + \dots \implies r = $ \makebox[1.5cm]{\dotfill} \quad [ \ ] ลู่เข้า (Convergent) \quad [ \ ] ลู่ออก (Divergent)
        \end{enumerate}
  \item \textbf{ผลรวมอนุกรมเรขาคณิตอนันต์ ($S_\infty = \frac{a_1}{1 - r}$ เมื่อ $|r| < 1$):}
        \begin{enumerate}[label=(\alph*),itemsep=2pt]
          \item จงหาผลรวมอนันต์ของ $12 + 6 + 3 + 1.5 + \dots$ ($a_1 = 12, r = 0.5$)\\[0.1em]
                วิธีทำ: $S_\infty = \frac{a_1}{1 - r} =$ \makebox[4cm]{\dotfill} $= $ \makebox[2cm]{\dotfill}
          \item จงแปลงทศนิยมซ้ำ $0.333\dots = \frac{3}{10} + \frac{3}{100} + \dots$ เป็นเศษส่วนอย่างต่ำ ($a_1 = 0.3, r = 0.1$)\\[0.1em]
                วิธีทำ: $S_\infty = \frac{0.3}{1 - 0.1} =$ \makebox[4cm]{\dotfill} $= $ \makebox[2cm]{\dotfill}
        \end{enumerate}
\end{enumerate}

\noindent\textbf{กิจกรรมที่ 4 : ฟังก์ชันเวียนเกิดในชีวิตประจำวัน (Recursive Functions in Daily Life)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item \textbf{โครงสร้าง 2 ส่วนสำคัญ:} ฟังก์ชันเวียนเกิดประกอบด้วย 1) \makebox[3.5cm]{\dotfill} และ 2) \makebox[3.5cm]{\dotfill}\\
        หากไม่มีกรณีฐาน (Base Case) จะเกิดผลเสียอย่างไร: \makebox[9cm]{\dotfill}
  \item \textbf{แบบจำลองดอกเบี้ยเงินฝากทบต้นรายปี ($10\%$ ต่อปี เงินต้น 10,000 บาท):}\\
        กรณีฐาน: $B(0) =$ \makebox[2.5cm]{\dotfill} \qquad ขั้นตอนเวียนเกิด: $B(n) =$ \makebox[4cm]{\dotfill}\\
        จงคำนวณ: $B(1) =$ \makebox[2cm]{\dotfill} บาท, \quad $B(2) =$ \makebox[2cm]{\dotfill} บาท, \quad $B(3) =$ \makebox[2cm]{\dotfill} บาท
  \item \textbf{แฟกทอเรียล ($n!$) กับการจัดลำดับสิ่งของ:} นิยาม $n! = n \times (n-1)!$ โดยมี Base Case $1! = 1$\\
        จงคลี่ขั้นตอนคำนวณ $4! = 4 \times 3! = 4 \times (3 \times 2!) = $ \makebox[4cm]{\dotfill} $= $ \makebox[1.5cm]{\dotfill}\\
        (หากมีคน 4 คน เข้าแถวซื้อของ จะจัดเรียงลำดับคิวได้ทั้งหมด \makebox[2cm]{\dotfill} แบบ)
  \item \textbf{ลำดับฟีโบนัชชีในธรรมชาติ ($F_n = F_{n-1} + F_{n-2}$ โดย $F_0 = 0, F_1 = 1$):}\\
        จงหาค่า: $F_2 = 1, \ F_3 = 2, \ F_4 = $ \makebox[1.2cm]{\dotfill}, \ $F_5 = $ \makebox[1.2cm]{\dotfill}, \ $F_6 = $ \makebox[1.2cm]{\dotfill}\\
        ยกตัวอย่างที่พบสัดส่วนฟีโบนัชชีในธรรมชาติ 1 ตัวอย่าง: \makebox[6cm]{\dotfill}
\end{enumerate}
