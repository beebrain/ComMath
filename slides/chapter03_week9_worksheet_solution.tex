%% ---------- กิจกรรม 1 ----------
\noindent\textbf{กิจกรรมที่ 1 : ลำดับเลขคณิตและเรขาคณิต (Arithmetic \& Geometric Sequences)} \hfill \textit{(6 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item กำหนดลำดับเลขคณิต $5, 8, 11, 14, \dots$ จงหา:
        \begin{enumerate}[label=(\alph*),itemsep=1pt]
          \item ผลต่างร่วม ($d$) = \ans{$8 - 5 = 3$}
          \item พจน์ทั่วไป ($a_n$) = $a_1 + (n-1)d = 5 + (n-1)(3) =$ \ans{$3n + 2$}
          \item พจน์ที่ 20 ($a_{20}$) = $3(20) + 2 =$ \ans{$62$}
        \end{enumerate}
  \item กำหนดลำดับเรขาคณิต $2, 6, 18, 54, \dots$ จงหา:
        \begin{enumerate}[label=(\alph*),itemsep=1pt]
          \item อัตราส่วนร่วม ($r$) = \ans{$\frac{6}{2} = 3$}
          \item พจน์ทั่วไป ($a_n$) = $a_1 \cdot r^{n-1} =$ \ans{$2 \cdot 3^{n-1}$}
          \item พจน์ที่ 8 ($a_8$) = $2 \cdot 3^{8-1} = 2 \cdot 3^7 = 2(2,187) =$ \ans{$4,374$}
        \end{enumerate}
\end{enumerate}

%% ---------- กิจกรรม 2 ----------
\noindent\textbf{กิจกรรมที่ 2 : อนุกรมกับการวิเคราะห์อัลกอริทึม (Series \& Loop Complexity)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item สำหรับ Nested Loop สองชั้น:
\begin{verbatim}
for i in range(1, n + 1):
    for j in range(1, i + 1):
        do_something() # c units of work
\end{verbatim}
        ผลบวกของรอบการทำงานทั้งหมดคือ $S_n = \sum_{i=1}^n i = 1 + 2 + \dots + n = $ \ans{$\frac{n(n+1)}{2} = \frac{1}{2}n^2 + \frac{1}{2}n$}\\
        ระบุ Big-O Time Complexity ของอัลกอริทึมนี้: \ans{$O(n^2)$}
  \item ในอัลกอริทึม Bubble Sort ข้อมูล $n$ ตัว รอบแรกเปรียบเทียบ $n-1$ ครั้ง รอบถัดมา $n-2$ ครั้ง ไปเรื่อยๆ จนถึง 1 ครั้ง\\
        จำนวนครั้งการเปรียบเทียบทั้งหมดเมื่อ $n = 100$ คือ: \ans{$\frac{100 \times 99}{2} = 4,950\text{ ครั้ง}$}
  \item ใน Complete Binary Tree ความสูง $h = 5$ (ชั้นที่ 0 ถึง 5) มีจำนวนโหนดรวมตามสูตรอนุกรมเรขาคณิต\\
        $S_6 = \sum_{i=0}^5 2^i = 1 + 2 + 4 + 8 + 16 + 32 = \frac{1(2^6 - 1)}{2 - 1} = $ \ans{$63\text{ โหนด}$}
\end{enumerate}

%% ---------- กิจกรรม 3 ----------
\noindent\textbf{กิจกรรมที่ 3 : ความสัมพันธ์เวียนเกิดและการวิเคราะห์การเรียกซ้ำ (Recurrence Relations)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item กำหนดสมการเวียนเกิด $T(n) = 2T(n/2) + c \cdot n$ โดยที่ $T(1) = c$ (Merge Sort Recurrence) จงคำนวณ:
        \begin{align*}
          T(2) &= 2T(1) + 2c = 2(c) + 2c = \ans{4c} \\
          T(4) &= 2T(2) + 4c = 2(4c) + 4c = 8c + 4c = \ans{12c} \\
          T(8) &= 2T(4) + 8c = 2(12c) + 8c = 24c + 8c = \ans{32c}
        \end{align*}
  \item สรุปรูปทั่วไปของ $T(n) = $ \ans{$c \cdot n(\log_2 n + 1)$} $\implies$ Big-O Complexity = \ans{$O(n \log n)$}
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 4 ----------
\noindent\textbf{กิจกรรมที่ 4 : อนุกรมเรขาคณิตและการประยุกต์ในวิทยาการคอมพิวเตอร์ (Geometric Series \& CS Applications)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item \textbf{ผลรวมอนุกรมเรขาคณิตจำกัด:} กำหนดอนุกรม $3 + 6 + 12 + 24 + \dots$ จงหาผลรวม 6 พจน์แรก ($S_6$)\\[0.15em]
        วิธีทำ: $a_1 = 3, \ r = $ \ans{$2$} , \quad สูตร $S_6 = \frac{a_1(r^6 - 1)}{r - 1} =$ \ans{$\frac{3(2^6 - 1)}{2 - 1} = 3(63) =$} \ans{$189$}
  \item \textbf{ผลรวมอนุกรมเรขาคณิตอนันต์ ($S_\infty$):} จงหาผลรวมอนันต์ของ $12 + 6 + 3 + 1.5 + \dots$\\[0.15em]
        วิธีทำ: $a_1 = 12, \ r = $ \ans{$\frac{1}{2} = 0.5$} (เนื่องจาก $|r| < 1$ อนุกรมลู่เข้า) $\implies S_\infty = \frac{a_1}{1 - r} =$ \ans{$\frac{12}{1 - 0.5} = \frac{12}{0.5} =$} \ans{$24$}
  \item \textbf{การเติบโตแบบทวีคูณและการขยายอาร์เรย์ (Dynamic Array Doubling Strategy):}\\
        ในภาษาไพทอน เมื่ออาร์เรย์เต็มจะขยายขนาดเป็น 2 เท่าเสมอ ($1 \to 2 \to 4 \to 8 \to \dots \to N$) มีต้นทุนคัดลอกสะสมคือ $N - 1$\\
        ต้นทุนถัวเฉลี่ยต่อการเพิ่มข้อมูล 1 ตัว (Amortized Cost) $= \frac{N - 1}{N} \approx$ \ans{$1$} $\implies$ Big-O = \ans{$O(1)$}
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 5 ----------
\noindent\textbf{กิจกรรมที่ 5 : ความสัมพันธ์ลำดับบางส่วนและการจัดลำดับงาน (Partial Order \& Topological Sort)} \hfill \textit{(8 นาที)}\\
มีงาน 5 งาน $\{A, B, C, D, E\}$ โดยมีเงื่อนไขก่อนหลัง (Prerequisite Relation $R$):
$R = \{(A, B), (A, C), (B, D), (C, D), (D, E)\}$
\begin{enumerate}[label=\arabic*)]
  \item งานใดบ้างที่สามารถเริ่มต้นทำเป็นงานแรกได้ ($\text{In-degree} = 0$) $\Rightarrow$ \ans{งาน $A$ (เพราะไม่มีเงื่อนไขก่อนหน้า ชี้เข้า = 0)}
  \item จงหาลำดับการทำงานที่ถูกต้องตามเงื่อนไข (Topological Order) มา 1 ลำดับ:\\[0.15em]
        ลำดับการทำงาน: \ans{$A \longrightarrow B \longrightarrow C \longrightarrow D \longrightarrow E$} \quad (หรือ $A \to C \to B \to D \to E$)
  \item หากเพิ่มเงื่อนไข $(E, A)$ จะสามารถหา Topological Sort ได้หรือไม่ เพราะเหตุใด?\\[0.15em]
        \ans{ไม่สามารถหาได้ เพราะจะเกิดวงรอบ (Directed Cycle) เช่น $A \to B \to D \to E \to A$ ทำให้กราฟไม่เป็น DAG และเกิดภาวะ Deadlock}
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 6 ----------
\noindent\textbf{กิจกรรมที่ 6 : การแก้ปัญหาการชนกันในตารางแฮช (Hash Collision Resolution)} \hfill \textit{(8 นาที)}\\
ตารางแฮชขนาด $m = 7$ (ช่อง $0, 1, 2, 3, 4, 5, 6$) ใช้ฟังก์ชันแฮช $h(k) = k \bmod 7$ มีลำดับคีย์เข้าคือ $12, 19, 26, 33$
\begin{enumerate}[label=\arabic*)]
  \item \textbf{วิธี Linear Probing ($h(k, i) = (k \bmod 7 + i) \bmod 7$):} จงเติมคีย์ลงในช่องตาราง:
        \begin{center}
        \small
        \begin{tabular}{|c|c|c|c|c|c|c|c|}
        \hline
        \textbf{ช่องดัชนี} & 0 & 1 & 2 & 3 & 4 & 5 & 6 \\ \hline
        \textbf{คีย์ที่จัดเก็บ} & \ans{26} & \ans{33} & \ans{-} & \ans{-} & \ans{-} & \ans{12} & \ans{19} \\ \hline
        \end{tabular}
        \end{center}
  \item \textbf{วิธี Chaining (Linked List):} คีย์ $12, 19, 26, 33$ จะถูกนำไปต่อที่ช่องดัชนีใดในตาราง:\\[0.15em]
        ช่องที่เกิดการต่อเป็นลูกโซ่คือ ช่องที่ \ans{5} โดยมีโซ่ของข้อมูลคือ \ans{$12 \longrightarrow 19 \longrightarrow 26 \longrightarrow 33$} (หรือ $33 \to 26 \to 19 \to 12$)
\end{enumerate}
\vspace{0.2em}

%% ---------- กิจกรรม 7 ----------
\noindent\textbf{กิจกรรมที่ 7 : การสังเคราะห์องค์ความรู้บทที่ 3 (Summary Reflection)} \hfill \textit{(4 นาที)}\\
จงอธิบายความสัมพันธ์ระหว่าง \textbf{ผลคูณคาร์ทีเซียน} $\to$ \textbf{ความสัมพันธ์} $\to$ \textbf{ฟังก์ชัน} ว่ามีข้อกำหนดที่เข้มงวดขึ้นอย่างไรในแต่ละขั้น?\\[0.15em]
\ans{1. ผลคูณคาร์ทีเซียน ($A \times B$): เซตของคู่อันดับทั้งหมดที่เป็นไปได้ ไม่จำกัดเงื่อนไข ขนาด $|A| \times |B|$}\\
\ans{2. ความสัมพันธ์ ($R \subseteq A \times B$): สับเซตของ $A \times B$ คัดเลือกเฉพาะคู่ที่ตรงตามเงื่อนไข สมาชิกหน้าจับคู่หลังได้หลายตัว}\\
\ans{3. ฟังก์ชัน ($f: A \to B$): เข้มงวดที่สุด สมาชิกหน้าทุกตัวต้องมีคู่ (Total) และจับคู่ได้เพียงตัวเดียวเท่านั้น (Single-valued)}
