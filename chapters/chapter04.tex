\chapter{ระบบเลขฐาน (Number Bases)}
\label{ch:number-bases}

ระบบเลขฐาน คือวิธีเขียนและแทนจำนวนด้วยตัวเลขหลายหลักประกอบกัน โดยค่าของแต่ละตำแหน่งขึ้นอยู่กับฐานของระบบนั้น คอมพิวเตอร์ทำงานด้วยสัญญาณไฟฟ้าสองสถานะ (เปิดและปิด) จึงใช้ระบบเลขฐานสองเป็นภาษาพื้นฐาน การเข้าใจระบบเลขฐานจึงมีความจำเป็นต่อการศึกษาสถาปัตยกรรมคอมพิวเตอร์ การเขียนโปรแกรมระดับต่ำ และการเข้ารหัสข้อมูล~\parencite{mano2018digital}

\section{ระบบเลขฐานสิบ (Decimal System)}
\index{ระบบเลขฐานสิบ}
\label{sec:decimal-system}

\textbf{ระบบเลขตำแหน่ง (Positional Notation System)}\index{ระบบเลขตำแหน่ง} กำหนดค่าของเลขโดดแต่ละตัวตามตำแหน่งที่ปรากฏ ทำงานแบบเดียวกับมาตรวัดระยะทางของรถยนต์ (Odometer) คือเมื่อนับตัวเลขในหลักขวาสุดครบ 10 ค่า ตั้งแต่ $0$ ถึง $9$ หลักถัดไปทางซ้ายจะถูกทดขึ้น 1 ค่า มนุษย์ใช้ระบบเลขฐานสิบเป็นหลักในชีวิตประจำวันเนื่องจากนับตามจำนวนนิ้วมือทั้ง 10 นิ้วได้สะดวก

ในระบบเลขฐานสิบ ค่าของตัวเลขในแต่ละตำแหน่งจะถูกกำหนดโดยผลคูณของตัวเลขหลักนั้นกับค่าน้ำหนักประจำหลัก $10^n$ โดย $n$ คือลำดับตำแหน่งหลัก ส่งผลให้ตัวเลขเดิมเมื่ออยู่ต่างตำแหน่งกันมีมูลค่าไม่เท่ากัน เช่น เลข $5$ ในหลักหน่วยมีค่า $5\times10^0=5$ ขณะที่เลข $5$ ในหลักร้อยมีค่า $5\times10^2=500$ การเขียนผลบวกของค่าน้ำหนักแต่ละหลักในหัวข้อถัดไปทำให้เห็นโครงสร้างนี้ได้ชัดเจน

\subsection{ค่าประจำตำแหน่ง}
ในระบบเลขฐานสิบ ค่าของจำนวนเต็มและทศนิยมเกิดจากการบวกสะสมค่าน้ำหนัก $10^n$ โดยตำแหน่งจำนวนเต็มให้ $n \ge 0$ เริ่มนับจากหลักขวาสุด (หลักหน่วย $10^0=1$, หลักสิบ $10^1=10$, หลักร้อย $10^2=100$) ส่วนทศนิยมให้ $n < 0$ ลดลงไปทางขวา ($10^{-1}=0.1$, $10^{-2}=0.01$) การเข้าใจกระจายค่าน้ำหนักประจำหลักนี้เป็นรากฐานสำคัญสำหรับการแปลงและเข้าใจระบบเลขฐานในคอมพิวเตอร์~\parencite{mano2018digital}

\begin{thrm}[ค่าประจำตำแหน่งของระบบเลขฐานสิบ]
\label{thrm:decimal-place-value}
จำนวนใด ๆ ในระบบเลขฐานสิบสามารถเขียนในรูปผลบวกของค่าประจำตำแหน่งได้

$$d_n d_{n-1} \ldots d_1 d_0 . d_{-1} d_{-2} \ldots d_{-m} = \sum_{i=-m}^{n} d_i \times 10^i$$
โดยที่ $d_i$ คือเลขโดดในแต่ละตำแหน่ง ($0 \le d_i \le 9$) และ $10^i$ คือค่าประจำตำแหน่ง
\end{thrm}

\begin{proof}
การพิสูจน์ค่าประจำตำแหน่งในระบบเลขตำแหน่ง (Positional Notation) พิจารณาเป็นลำดับขั้นตอนดังนี้
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{กำหนดค่าตำแหน่ง} ให้จำนวนคือ $N = d_n d_{n-1} \ldots d_1 d_0.d_{-1} d_{-2} \ldots d_{-m}$ ในฐาน 10 โดยที่ $d_i$ คือเลขโดดในตำแหน่งที่ $i$
    \item \textbf{คูณน้ำหนักประจำหลัก} ตัวเลข $d_i$ แต่ละหลักจะมีค่าน้ำหนักเท่ากับกำลังของฐาน 10 นั่นคือ $d_i \times 10^i$
    \item \textbf{รวมผลบวกทุกตำแหน่ง} นำค่าน้ำหนักทุกหลักตั้งแต่จำนวนเต็มและส่วนทศนิยมมารวมกัน
    \[
    N = \sum_{i=-m}^{n} d_i \times 10^i = d_n \times 10^n + d_{n-1} \times 10^{n-1} + \cdots + d_0 \times 10^0 + d_{-1} \times 10^{-1} + \cdots + d_{-m} \times 10^{-m}
    \]
\end{enumerate}
ผลรวมที่ได้สอดคล้องตรงตามนิยามการแทนจำนวนในระบบเลขฐานสิบ
\end{proof}

\begin{ex}[การจำแนกค่าประจำหลักในระบบเลขฐานสิบ]
\label{ex:decimal-place-value}
จงแสดงการกระจายค่าประจำหลักของจำนวนเต็ม $4729$ และจำนวนทศนิยม $53.847$ ในระบบเลขฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
4729 &= (4\times10^3)+(7\times10^2)+(2\times10^1)+(9\times10^0)\\
&= 4000+700+20+9\\
53.847 &= (5\times10^1)+(3\times10^0)+(8\times10^{-1})+(4\times10^{-2})+(7\times10^{-3})\\
&= 50+3+0.8+0.04+0.007
\end{align*}
$\therefore 4729 = 4000+700+20+9$ และ $53.847 = 50+3+0.8+0.04+0.007$ \solhere
\end{solution}

\section{ระบบเลขฐานสอง (Binary System)}
\label{sec:binary-system}

ในทางวิศวกรรมฮาร์ดแวร์ วงจรอิเล็กทรอนิกส์และไมโครโปรเซสเซอร์ของเครื่องคอมพิวเตอร์ไม่ได้ถูกสร้างมาให้รับรู้นิ้วมือ 10 นิ้วของมนุษย์ แต่ถูกสร้างขึ้นจากทรานซิสเตอร์นับล้านตัวที่ทำงานด้วย \textbf{สภาวะทางกายภาพเพียง 2 สถานะ} ได้แก่ ``มีกระแสไฟฟ้า/แรงดันสูง (ON / High Voltage)'' แทนด้วยสัญลักษณ์ $1$ และ ``ไม่มีกระแสไฟฟ้า/แรงดันต่ำ (OFF / Low Voltage)'' แทนด้วยสัญลักษณ์ $0$~\parencite{patterson2020organization}

ข้อจำกัดทางกายภาพนี้ส่งผลให้คอมพิวเตอร์จำเป็นต้องใช้ \textbf{ระบบเลขฐานสอง (Binary System)}\index{ระบบเลขฐานสอง} เป็นภาษาพื้นฐานในการจัดเก็บข้อมูล คำนวณ และประมวลผลคำสั่งทุกระดับ โดยสัญลักษณ์แต่ละหลักเรียกว่า \textbf{บิต (Bit)}\index{บิต} ซึ่งย่อมาจาก Binary Digit และมีค่าน้ำหนักประจำหลักเป็นกำลังของฐาน 2 ($2^0=1, 2^1=2, 2^2=4, 2^3=8, \dots$) ดังแสดงตัวอย่างการแปลงในรูปที่~\ref{fig:binary-place-value}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.86\textwidth]{Figures/png_ready/binary-conversion-labeled.png}
    \caption{การแปลง $1011_2$ เป็น $11_{10}$ ด้วยค่าน้ำหนักประจำหลัก}
    \label{fig:binary-place-value}
\end{figure}

รูปที่~\ref{fig:binary-place-value} แสดงบิต $1011_2$ พร้อมค่าน้ำหนัก $8,4,2,1$ ใต้แต่ละตำแหน่ง บิตสามตำแหน่งที่เป็น 1 ให้ค่า $1\times8$, $1\times2$ และ $1\times1$ ส่วนบิตตำแหน่งน้ำหนัก 4 เป็น 0 จึงให้ค่าเป็นศูนย์ ดังนั้น $(1011)_2=8+0+2+1=11_{10}$

\begin{principle}
\label{principle:binary-expansion}
จำนวนในระบบเลขฐานสองสามารถเขียนในรูปผลบวกของ $2^i$ ได้

$$(b_n b_{n-1} \ldots b_1 b_0)_2 = \sum_{i=0}^{n} b_i \times 2^i$$
โดยที่ $b_i \in \{0, 1\}$ แทนเลขโดดแต่ละหลักในฐานสอง
\end{principle}

\begin{rem}
\label{rem:binary-reading}
ข้อควรระวังคือ เลขฐานสอง $(1101)_2$ ต้องอ่านเรียงตัวว่า ``หนึ่ง-หนึ่ง-ศูนย์-หนึ่ง-ฐานสอง'' ห้ามอ่านว่า ``พันหนึ่งร้อยเอ็ด'' เนื่องจากแต่ละหลักไม่ได้มีค่าน้ำหนักเป็น $10^n$
\end{rem}

\begin{ex}[การแปลงเลขฐานสองจำนวนเต็มเป็นฐานสิบ]
\label{ex:bin-to-dec-1101}
แปลง $(1101)_2$ เป็นฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
(1101)_2 &= (1\times2^3)+(1\times2^2)+(0\times2^1)+(1\times2^0)\\
&= 8 + 4 + 0 + 1 = 13
\end{align*}
$\therefore (1101)_2 = (13)_{10}$ \solhere
\end{solution}

\begin{ex}[การแปลงเลขฐานสองห้าบิตเป็นฐานสิบ]
\label{ex:bin-to-dec-10110}
แปลง $(10110)_2$ เป็นฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
(10110)_2 &= (1\times2^4)+(0\times2^3)+(1\times2^2)+(1\times2^1)+(0\times2^0)\\
&= 16 + 0 + 4 + 2 + 0 = 22
\end{align*}
$\therefore (10110)_2 = (22)_{10}$ \solhere
\end{solution}

\begin{principle}
\label{principle:binary-fractional}
ส่วนทศนิยมในฐานสองมีค่าตามสมการ

$$(0.b_1 b_2 \ldots b_m)_2 = \sum_{i=1}^{m} b_i \times 2^{-i}$$
โดยที่ $b_i \in \{0, 1\}$ แทนเลขโดดในส่วนทศนิยมของฐานสอง ($2^{-1}=0.5, 2^{-2}=0.25, 2^{-3}=0.125$)
\end{principle}

\begin{ex}[การแปลงทศนิยมฐานสองเป็นฐานสิบ]
\label{ex:bin-frac-to-dec}
แปลง $(0.101)_2$ เป็นฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
(0.101)_2 &= (1\times2^{-1})+(0\times2^{-2})+(1\times2^{-3})\\
&= \frac{1}{2} + 0 + \frac{1}{8} = 0.5 + 0.125 = 0.625
\end{align*}
$\therefore (0.101)_2 = (0.625)_{10}$ \solhere
\end{solution}

\begin{ex}[การแปลงเลขฐานสองที่มีทั้งจำนวนเต็มและทศนิยม]
\label{ex:bin-mixed-to-dec}
แปลง $(10.011)_2$ เป็นฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
(10.011)_2 &= (1\times2^1)+(0\times2^0)+(0\times2^{-1})+(1\times2^{-2})+(1\times2^{-3})\\
&= 2 + 0 + 0 + 0.25 + 0.125 = 2.375
\end{align*}
$\therefore (10.011)_2 = (2.375)_{10}$ \solhere
\end{solution}

\section{ระบบเลขฐานแปด (Octal System)}
\index{ระบบเลขฐานแปด}
\label{sec:octal-system}

การเขียนและอ่านสายบิตเลขฐานสองที่ยาวมาก เช่น $110101100110_2$ ทำได้ยากและเกิดความผิดพลาดได้ง่าย วิธีลดความยาวที่ใช้กันคือการจับบิตเป็นกลุ่มขนาดเท่ากัน แล้วแทนแต่ละกลุ่มด้วยเลขโดดของฐานที่ใหญ่กว่าเพียงตัวเดียว ฐานที่ใช้ได้ต้องเป็นกำลังของสอง เพราะจะทำให้กลุ่มบิตขนาดคงที่แทนเลขโดดหนึ่งตัวได้พอดีโดยไม่มีเศษ ฐานแปดและฐานสิบหกจึงเป็นสองฐานที่ใช้กันแพร่หลายในงานคอมพิวเตอร์~\parencite{wakerly2018digital}

\textbf{ระบบเลขฐานแปด (Octal System)} ใช้หลักการนี้โดยจับบิตทีละ 3 ตัว เนื่องจาก $2^3 = 8$ บิตฐานสอง 3 บิตใด ๆ ตั้งแต่ $000_2$ ถึง $111_2$ จึงแทนค่าได้พอดีด้วยเลขโดดฐานแปดหนึ่งตัวตั้งแต่ $0$ ถึง $7$ การบันทึกรหัสบิตด้วยฐานแปดจึงสั้นลงเหลือหนึ่งในสามของความยาวเดิม ซึ่งเป็นเหตุผลที่คอมพิวเตอร์ยุคแรกนิยมใช้ฐานนี้

\begin{principle}
\label{principle:binary-octal-relation}
กลุ่มเลขฐานสอง 3 บิต แทนด้วยเลขโดดฐานแปด 1 ตัวได้พอดี เนื่องจาก $2^3 = 8$
\end{principle}

\begin{ex}[การแปลงฐานสองเป็นฐานแปดด้วยการจับกลุ่มบิต]
\label{ex:bin-to-oct-group}
แปลง $(110101)_2$ เป็นฐานแปด
\end{ex}
\begin{solution}
\begin{align*}
(110101)_2 &= \underbrace{110}_{6}\ \underbrace{101}_{5}\\
110_2 &= (1\times4)+(1\times2)+(0\times1) = 6\\
101_2 &= (1\times4)+(0\times2)+(1\times1) = 5
\end{align*}
$\therefore (110101)_2 = (65)_8$ \solhere
\end{solution}

\begin{ex}[การแปลงฐานแปดเป็นฐานสิบ]
\label{ex:oct-to-dec-472}
แปลง $(472)_8$ เป็นฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
(472)_8 &= (4\times8^2)+(7\times8^1)+(2\times8^0)\\
&= (4 \times 64) + (7 \times 8) + (2 \times 1) = 256 + 56 + 2 = 314
\end{align*}
$\therefore (472)_8 = (314)_{10}$ \solhere
\end{solution}

\begin{rem}
\label{rem:octal-history}
ฐานแปดเคยถูกใช้อย่างแพร่หลายในคอมพิวเตอร์ยุคแรก (เช่น เครื่อง PDP-8 ที่ใช้ขนาดคำ 12 บิต หรือ 36 บิต) ปัจจุบันในระบบคอมพิวเตอร์สมัยใหม่นิยมใช้ระบบเลขฐานสิบหกเป็นหลัก
\end{rem}

\section{ระบบเลขฐานสิบหก (Hexadecimal System)}
\index{ระบบเลขฐานสิบหก}
\label{sec:hexadecimal-system}

ในสถาปัตยกรรมคอมพิวเตอร์สมัยใหม่ หน่วยความจำและสถาปัตยกรรมโปรเซสเซอร์ถูกออกแบบให้ทำงานบนพื้นฐานของ \textbf{ไบต์ (Byte)}\index{ไบต์} ซึ่งมีขนาด 8 บิต และ \textbf{นิบเบิล (Nibble)}\index{นิบเบิล} ซึ่งมีขนาด 4 บิต \textbf{ระบบเลขฐานสิบหก (Hexadecimal System)} จึงถูกใช้เป็นสัญกรณ์ย่อมาตรฐานสำหรับเขียนแทนค่าของบิตเหล่านี้ในงานพัฒนาซอฟต์แวร์~\parencite{tremblay1975structures}

เนื่องจาก $2^4 = 16$ หมายความว่ากลุ่มบิต 4 บิตใด ๆ ($0000_2$ ถึง $1111_2$) สามารถแทนได้อย่างพอดีด้วยสัญลักษณ์ฐานสิบหก 1 ตัว ระบบนี้ใช้สัญลักษณ์ 16 ตัว ได้แก่ เลขโดด $0-9$ และตัวอักษรภาษาอังกฤษ $A-F$ (โดย $A=10, B=11, C=12, D=13, E=14, F=15$) ส่งผลให้ข้อมูล 1 ไบต์ (8 บิต) ย่อเหลือตัวเลขฐานสิบหกเพียง 2 หลักเท่านั้น เช่น รหัสสีบนเว็บ (Web Color) หรือตำแหน่งที่อยู่หน่วยความจำ (Memory Address) ดังแสดงในรูปที่~\ref{fig:hex-grouping}

\begin{principle}
\label{principle:binary-hex-relation}
ความสัมพันธ์ระหว่างฐานสองและฐานสิบหกคือ กลุ่มเลขโดดฐานสอง 4 ตัว หรือหนึ่งนิบเบิล แทนด้วยเลขโดดฐานสิบหก 1 ตัวได้พอดี เนื่องจาก $2^4 = 16$
โดยที่แต่ละกลุ่ม 4 บิต มีค่าตั้งแต่ $0000_2 = 0$ ถึง $1111_2 = 15 = \mathrm{F}_{16}$
\end{principle}

\begin{ex}[การแปลงฐานสองเป็นฐานสิบหกด้วยการจับกลุ่มบิต]
\label{ex:bin-to-hex-group}
แปลง $(11010110)_2$ เป็นฐานสิบหก
\end{ex}

(11010110)_2 &= \underbrace{1101}_{\mathrm{D}}\ \underbrace{0110}_{6}\\

\begin{ex}[การแปลงฐานสิบหกที่มีทศนิยมเป็นฐานสิบ]
\label{ex:hex-frac-to-dec}

\end{ex}

(3\mathrm{A}.\mathrm{F})_{16} &= (3\times16^1)+(10\times16^0)+(15\times16^{-1}) \quad (\mathrm{A}=10,\ \mathrm{F}=15)\\
&= 48 + 10 + \frac{15}{16} = 58 + 0.9375 = 58.9375
\end{align*}
$\therefore (3\mathrm{A}.\mathrm{F})_{16} = (58.9375)_{10}$ \solhere
\end{solution}


\section{การแปลงฐานทั่วไป}
\label{sec:general-base-conversion}

ในการสื่อสารระหว่างมนุษย์กับระบบคอมพิวเตอร์ หรือระหว่างระบบประมวลผลต่างชนิดกัน ข้อมูลจำนวนเดียวกันต้องถูกแปลความหมายสลับไปมาระหว่างฐานต่าง ๆ จุดร่วมของการแปลงทุกแบบคือใช้ฐานสิบเป็นตัวกลาง เพราะเป็นฐานที่มนุษย์คำนวณได้คล่องที่สุด ยกเว้นการแปลงระหว่างฐานที่เป็นกำลังของสองด้วยกันเอง ซึ่งใช้การจับกลุ่มบิตได้โดยตรงและเร็วกว่ามาก การแปลงฐานแบ่งออกเป็น 2 ทิศทางหลักดังนี้~\parencite{kolman2008discrete}
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{การแปลงจากฐานใด ๆ เป็นฐานสิบ} คูณเลขโดดในแต่ละหลักด้วยค่าน้ำหนักประจำหลักนั้น ($b^i$) แล้วรวมผลคูณทั้งหมด
    \item \textbf{การแปลงจากฐานสิบเป็นฐานใด ๆ} หารจำนวนด้วยฐานปลายทางอย่างต่อเนื่อง บันทึกเศษของแต่ละรอบ แล้วอ่านเศษจากล่างขึ้นบนเป็นผลลัพธ์
\end{enumerate}

\subsection{การแปลงจากฐานใด ๆ เป็นฐานสิบ}
\label{ssec:any-base-to-decimal}
การแปลงจากฐาน $b$ ใด ๆ เป็นฐานสิบทำได้โดยนำเลขโดดในแต่ละตำแหน่งคูณกับค่าน้ำหนัก $b^i$ แล้วนำผลคูณทุกตำแหน่งมารวมกัน วิธีนี้เป็นการอ่านความหมายของระบบเลขตำแหน่งตรง ๆ โดยตำแหน่งขวาสุดของส่วนจำนวนเต็มมีเลขชี้กำลังเป็นศูนย์ แล้วเพิ่มขึ้นทีละหนึ่งเมื่อเลื่อนไปทางซ้าย และลดลงทีละหนึ่งเมื่อเลื่อนไปทางขวาของจุดทศนิยม~\parencite{mano2018digital}

\begin{principle}
\label{principle:general-base-expansion}
ถ้า $(d_n d_{n-1} \ldots d_1 d_0)_b$ เป็นจำนวนในฐาน $b$ จะมีค่าเท่ากับ

$$(d_n d_{n-1} \ldots d_1 d_0)_b = \sum_{i=0}^{n} d_i \times b^i$$
โดยที่ $d_i$ คือเลขโดดในแต่ละตำแหน่ง และ $b$ คือฐานของระบบเลข
\end{principle}

\begin{ex}[การแปลงจากฐานใด ๆ เป็นฐานสิบด้วยผลบวกถ่วงน้ำหนัก]
\label{ex:any-base-to-dec}
แปลง $(2\mathrm{B}3)_{16}$ เป็นฐานสิบ และ $(57)_8$ เป็นฐานสิบ
\end{ex}
\begin{solution}
\begin{align*}
(2\mathrm{B}3)_{16} &= (2\times16^2)+(11\times16^1)+(3\times16^0) \quad (\mathrm{B}=11)\\
&= 512 + 176 + 3 = 691\\
(57)_8 &= (5\times8^1)+(7\times8^0) = 40 + 7 = 47
\end{align*}
$\therefore (2\mathrm{B}3)_{16} = (691)_{10}$ และ $(57)_8 = (47)_{10}$ \solhere
\end{solution}

\subsection{การแปลงจากฐานสิบเป็นฐานใด ๆ}
\label{ssec:decimal-to-any-base}
การแปลงจำนวนเต็มจากฐานสิบเป็นฐาน $b$ ใช้วิธีหารต่อเนื่อง (Successive Division) โดยนำฐาน $b$ ไปหารจำนวนตั้งต้นและผลหารในแต่ละรอบ จดเศษที่เหลือไว้ แล้วรวบรวมเศษอ่านจากล่างขึ้นบน เหตุผลที่ต้องอ่านย้อนกลับคือเศษที่ได้จากการหารรอบแรกเป็นเลขโดดของหลักที่มีค่าน้ำหนักต่ำสุด ส่วนเศษจากรอบสุดท้ายเป็นเลขโดดของหลักที่มีค่าน้ำหนักสูงสุด การหารจะสิ้นสุดเมื่อผลหารเป็นศูนย์~\parencite{patterson2020organization}

\begin{principle}
\label{principle:decimal-to-any-base-steps}
ขั้นตอนการแปลงจำนวนเต็มจากฐานสิบเป็นฐาน $b$ มีดังนี้
\begin{enumerate}
    \item หารจำนวนตั้งต้นด้วย $b$ บันทึกผลหารและเศษ
    \item นำผลหารในข้อ 1 มาหารด้วย $b$ ต่อไป จดเศษ
    \item ทำซ้ำจนกว่าผลหารจะเป็น 0
    \item รวบรวมเศษที่ได้อ่านย้อนกลับจากล่างขึ้นบนเป็นผลลัพธ์
\end{enumerate}
\end{principle}

\begin{ex}[การแปลงฐานสิบเป็นฐานสองด้วยการหารต่อเนื่อง]
\label{ex:dec-to-bin-45}
แปลง $45$ ฐานสิบเป็นฐานสอง
\end{ex}
\begin{solution}
\begin{align*}
45 \div 2 &= 22 \quad \text{เศษ } 1 \quad (\text{บิตนัยสำคัญน้อยสุด})\\
22 \div 2 &= 11 \quad \text{เศษ } 0\\
11 \div 2 &= 5  \quad \text{เศษ } 1\\
 5 \div 2 &= 2  \quad \text{เศษ } 1\\
 2 \div 2 &= 1  \quad \text{เศษ } 0\\
 1 \div 2 &= 0  \quad \text{เศษ } 1 \quad (\text{บิตนัยสำคัญมากสุด})
\end{align*}
อ่านเศษจากล่างขึ้นบน และตรวจสอบได้ว่า $(101101)_2 = 32+8+4+1 = 45$

$\therefore (45)_{10} = (101101)_2$ \solhere
\end{solution}

\begin{ex}[การแปลงฐานสิบเป็นฐานแปดด้วยการหารต่อเนื่อง]
\label{ex:dec-to-oct-158}
แปลง $158$ ฐานสิบเป็นฐานแปด
\end{ex}
\begin{solution}
\begin{align*}
158 \div 8 &= 19 \quad \text{เศษ } 6\\
 19 \div 8 &= 2  \quad \text{เศษ } 3\\
  2 \div 8 &= 0  \quad \text{เศษ } 2
\end{align*}
อ่านเศษจากล่างขึ้นบน และตรวจสอบได้ว่า $(236)_8 = (2 \times 64) + (3 \times 8) + 6 = 158$

$\therefore (158)_{10} = (236)_8$ \solhere
\end{solution}

\begin{ex}[การแปลงฐานสิบเป็นฐานสิบหกที่ต้องแปลงเศษเป็นอักษร]
\label{ex:dec-to-hex-2543}
แปลง $2543$ ฐานสิบเป็นฐานสิบหก
\end{ex}
\begin{solution}
\begin{align*}
2543 \div 16 &= 158 \quad \text{เศษ } 15 \implies \mathrm{F}\\
 158 \div 16 &= 9   \quad \text{เศษ } 14 \implies \mathrm{E}\\
   9 \div 16 &= 0   \quad \text{เศษ } 9  \implies 9
\end{align*}
อ่านเศษจากล่างขึ้นบน และตรวจสอบได้ว่า $(9\mathrm{EF})_{16} = 2304 + 224 + 15 = 2543$

$\therefore (2543)_{10} = (9\mathrm{EF})_{16}$ \solhere
\end{solution}

\section{การแปลงจำนวนทศนิยมระหว่างฐาน}
\label{sec:fractional-base-conversion}

ในการประมวลผลทางคอมพิวเตอร์ การแปลงส่วนทศนิยมระหว่างฐานซับซ้อนกว่าการแปลงจำนวนเต็ม วิธีที่ใช้คือ \textbf{วิธีคูณต่อเนื่อง (Successive Multiplication)} ซึ่งคูณส่วนทศนิยมด้วยฐานปลายทางซ้ำไปเรื่อย ๆ แล้วสกัดส่วนจำนวนเต็มที่ได้ออกมาเป็นเลขโดดทีละหลัก ข้อควรระวังคือทศนิยมที่สิ้นสุดในฐานสิบอาจกลายเป็นทศนิยมซ้ำไม่รู้จบในฐานสอง เช่น $0.1$ ในฐานสิบ ซึ่งเป็นสาเหตุที่การคำนวณทศนิยมในคอมพิวเตอร์มีความคลาดเคลื่อนสะสม~\parencite{wakerly2018digital}

จำนวนทศนิยมบางจำนวนแทนด้วยบิตจำนวนจำกัดไม่ได้ เช่น $0.1_{10}$ และ $0.7_{10}$ มีเลขโดดซ้ำไม่สิ้นสุดเมื่อเขียนในฐานสอง เมื่อระบบคอมพิวเตอร์จัดเก็บได้เพียงจำนวนบิตจำกัด จึงต้องตัดหรือปัดเศษบิตบางส่วน การตัดหรือปัดเศษนี้เป็นสาเหตุของความคลาดเคลื่อนในการคำนวณแบบจุดลอยตัว (Floating-Point Rounding Error)

\subsection{การแปลงส่วนทศนิยมจากฐานสิบเป็นฐานอื่น}
\label{ssec:fractional-conversion}
การแปลงส่วนทศนิยมใช้วิธีคูณต่อเนื่อง (Successive Multiplication) กับส่วนทศนิยมด้วยฐาน $b$ แล้วดึงส่วนจำนวนเต็มที่เป็นผลคูณออกมาบันทึกเป็นหลักทศนิยม ต่างจากการแปลงจำนวนเต็มตรงที่อ่านผลลัพธ์จากบนลงล่าง เพราะหลักที่ได้จากการคูณรอบแรกอยู่ถัดจากจุดทศนิยมทันที การคูณจะสิ้นสุดเมื่อส่วนทศนิยมเหลือศูนย์ หรือเมื่อครบจำนวนหลักที่ต้องการในกรณีที่เกิดการวนซ้ำ~\parencite{tremblay1975structures}

\begin{principle}
\label{principle:fractional-conversion-steps}
ขั้นตอนการแปลงส่วนทศนิยมจากฐานสิบเป็นฐาน $b$ มีดังนี้
\begin{enumerate}
    \item นำส่วนทศนิยมคูณด้วยฐาน $b$
    \item บันทึกส่วนจำนวนเต็มของผลคูณเป็นเลขโดดทศนิยมหลักถัดไป
    \item นำเฉพาะส่วนทศนิยมที่เหลือไปคูณด้วย $b$ ต่อไป
    \item ทำซ้ำจนกว่าส่วนทศนิยมจะเป็น 0 หรือได้จำนวนหลักความแม่นยำตามต้องการ
    \item อ่านเลขโดดที่บันทึกไว้เรียงจากบนลงล่าง
\end{enumerate}
\end{principle}

\begin{ex}[การแปลงทศนิยมที่สิ้นสุดเป็นฐานสอง]
\label{ex:frac-terminating}
แปลง $0.625$ ฐานสิบเป็นฐานสอง
\end{ex}
\begin{solution}
\begin{align*}
0.625 \times 2 &= 1.25 \quad \implies \text{หลัก } 1 \quad (\text{เหลือ } 0.25)\\
0.25  \times 2 &= 0.5  \quad \implies \text{หลัก } 0 \quad (\text{เหลือ } 0.5)\\
0.5   \times 2 &= 1.0  \quad \implies \text{หลัก } 1 \quad (\text{เหลือ } 0.0)
\end{align*}
อ่านหลักจากบนลงล่าง และตรวจสอบได้ว่า $(0.101)_2 = 0.5 + 0.125 = 0.625$

$\therefore (0.625)_{10} = (0.101)_2$ \solhere
\end{solution}

\begin{ex}[การแปลงทศนิยมที่ซ้ำไม่รู้จบเป็นฐานสอง]
\label{ex:frac-repeating}
แปลง $0.7$ ฐานสิบเป็นฐานสอง (แสดง 8 หลัก)
\end{ex}
\begin{solution}
\begin{align*}
0.7 \times 2 &= 1.4 \quad \implies \text{หลัก } 1\\
0.4 \times 2 &= 0.8 \quad \implies \text{หลัก } 0\\
0.8 \times 2 &= 1.6 \quad \implies \text{หลัก } 1\\
0.6 \times 2 &= 1.2 \quad \implies \text{หลัก } 1\\
0.2 \times 2 &= 0.4 \quad \implies \text{หลัก } 0\\
0.4 \times 2 &= 0.8 \quad \implies \text{หลัก } 0\\
0.8 \times 2 &= 1.6 \quad \implies \text{หลัก } 1\\
0.6 \times 2 &= 1.2 \quad \implies \text{หลัก } 1
\end{align*}
ส่วนทศนิยม $0.4$ วนกลับมาซ้ำ ทำให้ได้ทศนิยมซ้ำไม่รู้จบ $(0.7)_{10} = (0.1\overline{0110})_2$

$\therefore (0.7)_{10} \approx (0.10110011)_2$ เมื่อแสดงผล 8 หลัก \solhere
\end{solution}

\begin{ex}[การแปลงทศนิยมฐานสิบเป็นฐานแปด]
\label{ex:frac-to-oct}
แปลง $0.3125$ ฐานสิบเป็นฐานแปด
\end{ex}
\begin{solution}
\begin{align*}
0.3125 \times 8 &= 2.5 \quad \implies \text{หลัก } 2 \quad (\text{เหลือ } 0.5)\\
0.5    \times 8 &= 4.0 \quad \implies \text{หลัก } 4 \quad (\text{เหลือ } 0.0)
\end{align*}
อ่านหลักจากบนลงล่าง และตรวจสอบได้ว่า $(0.24)_8 = \frac{2}{8} + \frac{4}{64} = 0.3125$

$\therefore (0.3125)_{10} = (0.24)_8$ \solhere
\end{solution}

\section{การบวกและลบเลขฐานสอง}
\label{sec:binary-add-sub}

การบวกและการลบในระบบเลขฐานสองเป็นการดำเนินการที่หน่วยคำนวณและตรรกะ (Arithmetic Logic Unit -- ALU) ภายในหน่วยประมวลผลกลางดำเนินการกับบิตโดยตรง~\parencite{kolman2008discrete}

หัวข้อนี้เริ่มจากการคำนวณด้วยมือโดยใช้การทดและการยืมเช่นเดียวกับฐานสิบ ความสัมพันธ์ระหว่างการบวก การลบ และวงจรตรรกะจะศึกษาเพิ่มเติมเมื่อกล่าวถึงการแทนจำนวนติดลบ

กฎการบวกและการลบเลขฐานสองใช้หลักการเดียวกับฐานสิบ แต่ง่ายกว่ามากเพราะเลขโดดมีเพียง $0$ และ $1$ ดังนี้
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{บิตทด (Carry Bit)}\index{บิตทด} เกิดขึ้นเมื่อบวกกันได้ $1+1=10_2$ (เป็น $0$ แล้วทด $1$ ไปยังหลักซ้ายมือที่มีค่าน้ำหนักเป็น $2$ เท่า)
    \item \textbf{บิตยืม (Borrow Bit)}\index{บิตยืม} เกิดขึ้นเมื่อตัวตั้งน้อยกว่าตัวลบ ($0-1$) ต้องยืมค่าน้ำหนัก $2$ จากหลักทางซ้ายมือเข้ามาช่วยลบ
\end{enumerate}

\subsection{การบวกเลขฐานสอง}
\index{การบวกเลขฐานสอง}
การบวกเลขฐานสองใช้หลักการเดียวกับการบวกเลขฐานสิบ คือบวกทีละหลักจากขวาไปซ้ายพร้อมส่งบิตทดไปยังหลักถัดไป ต่างกันที่การทดเกิดขึ้นเมื่อผลรวมถึง 2 แทนที่จะเป็น 10 กรณีที่ต้องจำมีเพียงห้ากรณี เพราะเลขโดดมีแค่สองค่า การบวกเลขฐานสองมีกฎพื้นฐาน 5 ประการดังนี้~\parencite{mano2018digital}

\begin{law}
\label{law:binary-addition-rules}
การบวกเลขฐานสอง
\begin{enumerate}
    \item $0 + 0 = 0$
    \item $0 + 1 = 1$
    \item $1 + 0 = 1$
    \item $1 + 1 = 10_2$ (ใส่ 0 ทด 1 ไปหลักซ้าย)
    \item $1 + 1 + 1 = 11_2$ (ใส่ 1 ทด 1 ไปหลักซ้าย)
\end{enumerate}
\end{law}

\begin{ex}[การบวกเลขฐานสองที่เกิดบิตทดทุกหลัก]
\label{ex:bin-add-carry-all}
บวก $(1011)_2$ กับ $(1101)_2$ ได้ดังนี้
\end{ex}
\begin{solution}
\[
\begin{array}{r}
1011 \\
+\,1101 \\
\hline
11000
\end{array}
\]
ตรวจสอบได้ว่า $11_{10} + 13_{10} = 24_{10} = (11000)_2$

$\therefore (1011)_2 + (1101)_2 = (11000)_2$ \solhere
\end{solution}

\begin{ex}[การบวกเลขฐานสองที่ผลลัพธ์ล้นจำนวนบิต]
\label{ex:bin-add-overflow}
บวก $(1111)_2$ กับ $(0001)_2$ ได้ดังนี้
\end{ex}
\begin{solution}
\[
\begin{array}{r}
1111 \\
+\,0001 \\
\hline
10000
\end{array}
\]
ตรวจสอบได้ว่า $15_{10} + 1_{10} = 16_{10} = (10000)_2$

$\therefore (1111)_2 + (0001)_2 = (10000)_2$ \solhere
\end{solution}

\subsection{การลบเลขฐานสอง}
\index{การลบเลขฐานสอง}
\label{ssec:binary-subtraction}
การลบเลขฐานสองโดยตรงอาศัยการยืมค่าน้ำหนัก $2$ จากหลักซ้ายมือเมื่อพบกรณี $0-1$ ซึ่งเป็นหลักการเดียวกับการยืมค่าน้ำหนัก $10$ ในฐานสิบ อย่างไรก็ตาม วงจรจริงในคอมพิวเตอร์ไม่ได้สร้างวงจรลบแยกต่างหาก แต่ใช้วิธีบวกด้วยคอมพลีเมนต์สองแทน ซึ่งจะศึกษาในหัวข้อจำนวนติดลบ~\parencite{patterson2020organization}

\begin{law}
\label{law:binary-subtraction-rules}
การลบเลขฐานสอง
\begin{enumerate}
    \item $0 - 0 = 0$
    \item $1 - 0 = 1$
    \item $1 - 1 = 0$
    \item $0 - 1 = 1$ (ยืม 1 จากหลักซ้าย ซึ่งมีค่าเป็น 2 ในหลักปัจจุบัน)
\end{enumerate}
\end{law}

\begin{ex}[การลบเลขฐานสองที่ต้องยืม]
\label{ex:bin-sub-borrow}
ลบ $(1100)_2$ ด้วย $(0101)_2$ ได้ดังนี้
\end{ex}
\begin{solution}
\[
\begin{array}{r}
1100 \\
-\,0101 \\
\hline
0111
\end{array}
\]
ตรวจสอบได้ว่า $12_{10} - 5_{10} = 7_{10} = (0111)_2$

$\therefore (1100)_2 - (0101)_2 = (0111)_2$ \solhere
\end{solution}

\begin{ex}[การลบเลขฐานสองที่ต้องยืมต่อเนื่องข้ามหลัก]
\label{ex:bin-sub-chain-borrow}
ลบ $(1000)_2$ ด้วย $(0001)_2$ (แสดงขั้นตอนการยืม)

\end{ex}

\begin{solution}
\[
\begin{array}{r}
1000 \\
-\,0001 \\
\hline
0111
\end{array}
\]
การยืมต่อเนื่องเกิดขึ้นดังนี้ หลักขวาสุด $0-1$ ต้องยืมจากหลักที่สี่ผ่านหลักที่สองและสาม ทำให้ $1000_2$ กลายเป็นค่าประจำหลัก $0,1,1,10_2$ ก่อนลบ

$\therefore (1000)_2 - (0001)_2 = (0111)_2$

ตรวจคำตอบได้จาก $(1000)_2 = 8_{10}$, $(0001)_2 = 1_{10}$, และ $8-1 = 7 = (0111)_2$ \solhere
\end{solution}

\section{การคูณและการหารเลขฐานสอง}
\label{sec:binary-mult-div}

การคูณเลขฐานสองใช้การเลื่อนบิตและการบวกผลคูณย่อย เพราะบิตของตัวคูณมีได้เพียง 0 หรือ 1 ส่วนการหารใช้การเปรียบเทียบค่าบิต การลบ และการนำบิตถัดไปลงมาตามลำดับ กฎพื้นฐานของการคูณมีดังนี้~\parencite{wakerly2018digital}
\begin{enumerate}[label=\textbf{\arabic*.}]
    \item เมื่อบิตตัวคูณเป็น $0$ ผลคูณย่อยจะเป็น $0$ ทั้งหมด (ไม่ต้องบวกเพิ่ม)
    \item เมื่อบิตตัวคูณเป็น $1$ ผลคูณย่อยจะเท่ากับตัวตั้งเดิมแต่ถูกเลื่อนตำแหน่งบิตไปทางซ้าย (Bit Shift)
\end{enumerate}
ฮาร์ดแวร์คอมพิวเตอร์จึงใช้เทคนิค \textbf{เลื่อนตำแหน่งบิตและบวกสะสม (Shift-and-Add)} เพื่อคำนวณผลคูณจากการเลื่อนบิตและการบวก

\subsection{การคูณเลขฐานสอง}
\index{การคูณเลขฐานสอง}
การคูณเลขฐานสองมีกฎพื้นฐาน 4 ประการที่ตรงกับตารางค่าความจริงของประตูตรรกะ AND พอดี เพราะผลคูณของสองบิตเป็น 1 ก็ต่อเมื่อบิตทั้งสองเป็น 1 ความสอดคล้องนี้ทำให้วงจรคูณสร้างขึ้นจากประตูตรรกะ AND ร่วมกับวงจรบวกได้โดยตรง โดยไม่ต้องออกแบบวงจรคูณเฉพาะขึ้นมาใหม่ ซึ่งเป็นข้อได้เปรียบที่ระบบเลขฐานสิบไม่มี~\parencite{tremblay1975structures}

\begin{law}
\label{law:binary-multiplication-rules}
การคูณเลขฐานสอง
\begin{enumerate}
    \item $0 \times 0 = 0$
    \item $0 \times 1 = 0$
    \item $1 \times 0 = 0$
    \item $1 \times 1 = 1$
\end{enumerate}
\end{law}

\begin{ex}[การคูณเลขฐานสองด้วยวิธีเลื่อนบิตและบวกสะสม]
\label{ex:bin-multiply}
คูณ $(1011)_2$ ด้วย $(1101)_2$ ได้ดังนี้
\end{ex}
\begin{solution}
\[
\begin{array}{r}
1011 \\
\times\,1101 \\
\hline
1011 \\
0000\phantom{0} \\
1011\phantom{00} \\
1011\phantom{000} \\
\hline
10001111
\end{array}
\]
ตรวจสอบได้ว่า $11_{10} \times 13_{10} = 143_{10} = (10001111)_2$

$\therefore (1011)_2 \times (1101)_2 = (10001111)_2$ \solhere
\end{solution}

\begin{rem}
\label{rem:shift-add}
วิธี Shift-and-Add สร้างผลคูณย่อยจากการเลื่อนบิต แล้วรวมผลคูณย่อยด้วยวงจรบวก
\end{rem}

\subsection{การหารเลขฐานสอง}
\index{การหารเลขฐานสอง}
การหารเลขฐานสองของจำนวนเต็มไม่ติดลบใช้ขั้นตอนเดียวกับการหารยาวในฐานสิบ เริ่มจากบิตทางซ้าย เปรียบเทียบค่าปัจจุบันกับตัวหาร ถ้าค่าปัจจุบันไม่น้อยกว่าตัวหาร ให้เขียนบิต $1$ ในผลหารแล้วลบตัวหารออก ถ้าน้อยกว่าตัวหาร ให้เขียนบิต $0$ แล้วนำบิตถัดไปลงมา ตัวหารต้องไม่เป็นศูนย์

\begin{ex}[การหารเลขฐานสองด้วยวิธีหารยาว]
\label{ex:bin-divide}
หาร $(11000)_2$ ด้วย $(100)_2$
\end{ex}
\begin{solution}
เริ่มพิจารณา $110_2$ ซึ่งมีค่าไม่น้อยกว่า $100_2$ จึงเขียนบิตแรกของผลหารเป็น $1$ แล้วลบ $100_2$ ออก
\begin{align*}
110_2 - 100_2 &= 10_2 &&(\text{ผลหารหลักแรกเป็น } 1)\\
100_2 - 100_2 &= 0_2 &&(\text{นำบิต } 0 \text{ ตัวถัดไปลงมา; ผลหารหลักถัดไปเป็น } 1)\\
0_2 &< 100_2 &&(\text{นำบิต } 0 \text{ สุดท้ายลงมา; ผลหารหลักสุดท้ายเป็น } 0)
\end{align*}
ดังนั้นผลหารคือ $(110)_2$ และเศษเป็น $0$ ตรวจสอบได้ว่า $(110)_2 \times (100)_2 = (11000)_2$

$\therefore (11000)_2 \div (100)_2 = (110)_2$ \solhere
\end{solution}


\section{การแทนข้อมูลในคอมพิวเตอร์}
\label{sec:data-representation}

เนื่องจากวงจรอิเล็กทรอนิกส์ในหน่วยประมวลผลและหน่วยความจำของคอมพิวเตอร์สามารถจัดเก็บและประมวลผลได้เฉพาะสัญญาณบิต $0$ และ $1$ เท่านั้น ข้อมูลทุกประเภทในโลกจริง เช่น ตัวอักษร สัญลักษณ์ ข้อความ รูปภาพ หรือเสียง จึงต้องถูกแปลงให้อยู่ในรูป \textbf{ตัวเลขรหัส (Numeric Code Points)} เสียก่อน~\parencite{kolman2008discrete}

การแทนข้อมูลตัวอักษรอาศัย \textbf{ตารางเทียบรหัสประจำตัวอักขระ (Character Encoding Table)} ซึ่งกำหนดตัวเลขประจำตัวให้อักษรแต่ละตัว เพื่อบันทึกลงดิสก์หรือหน่วยความจำในรูปเลขฐานสองหรือฐานสิบหก

\subsection{รหัสแอสกี (ASCII)}
\index{รหัสแอสกี}
\label{ssec:ascii-representation}
รหัส ASCII เป็นมาตรฐานแรกสุดที่นิยมใช้อย่างแพร่หลายในการแทนอักขระภาษาอังกฤษและตัวเลข โดยใช้พื้นที่จัดเก็บขนาด 7 บิต (หรือ 1 ไบต์ 8 บิต รวมบิตตรวจสอบ Parity) สามารถแทนสัญลักษณ์ได้ทั้งหมด 128 ตัวอักษร ($0$ ถึง $127$) การจัดเรียงรหัสไม่ได้สุ่ม แต่ออกแบบให้ตัวอักษรพิมพ์ใหญ่และพิมพ์เล็กต่างกันเพียงบิตเดียว และตัวเลข $0$ ถึง $9$ เรียงติดกัน ทำให้แปลงอักขระเป็นค่าตัวเลขได้ด้วยการลบอย่างง่าย~\parencite{mano2018digital}

\begin{principle}
\label{principle:ascii-ranges}
ตารางรหัสแอสกีแบ่งช่วงการแทนสัญลักษณ์หลักดังนี้
\begin{enumerate}
    \item ช่วง 0 ถึง 31 เป็นอักขระควบคุมระบบ (Control Characters) เช่น Line Feed, Enter และ Tab
    \item ช่วง 32 ถึง 47 เริ่มด้วยอักขระเว้นวรรคที่รหัส 32 ตามด้วยสัญลักษณ์พิเศษและเครื่องหมายวรรคตอน
    \item ช่วง 48 ถึง 57 เป็นเลขโดด `0' ถึง `9' โดยรหัสของ `0' คือ $48_{10} = 30_{16}$
    \item ช่วง 65 ถึง 90 เป็นอักษรภาษาอังกฤษพิมพ์ใหญ่ `A' ถึง `Z'
    \item ช่วง 97 ถึง 122 เป็นอักษรภาษาอังกฤษพิมพ์เล็ก `a' ถึง `z'
    \item ช่วงที่เหลือคือ 58 ถึง 64, 91 ถึง 96 และ 123 ถึง 127 เป็นสัญลักษณ์และวงเล็บเพิ่มเติม
\end{enumerate}
\end{principle}

\begin{table}[H]
\centering
\caption{ตาราง ASCII Code แสดงการเปรียบเทียบรหัสฐานสิบและฐานสิบหก}
\label{tab:ascii}
\begin{tabular}{|c|c|c|}
\hline
\textbf{อักขระ} & \textbf{ASCII (ฐานสิบ)} & \textbf{ASCII (ฐานสิบหก)} \\
\hline
'A' & 65 & $41_{16}$ \\
'B' & 66 & $42_{16}$ \\
'0' & 48 & $30_{16}$ \\
'/' & 47 & $2\mathrm{F}_{16}$ \\
'a' & 97 & $61_{16}$ \\
'\{' & 123 & $7\mathrm{B}_{16}$ \\
\hline
\end{tabular}
\end{table}

\begin{ex}[การเข้ารหัสข้อความด้วยรหัสแอสกี]
\label{ex:ascii-encode-hi}
จงแสดงรหัสบิตฐานสิบหกของคำว่า ``Hi'' ในระบบ ASCII
\end{ex}
\begin{solution}
\begin{align*}
\text{'H'} &= 72_{10} = (48)_{16} = \texttt{0x48}\\
\text{'i'} &= 105_{10} = (69)_{16} = \texttt{0x69}
\end{align*}
$\therefore$ ข้อความ ``Hi'' เข้ารหัสได้เป็น \texttt{0x4869} \solhere
\end{solution}

\begin{rem}
\label{rem:ascii-limitation}
ข้อจำกัดสำคัญของ ASCII คือสามารถแทนอักขระได้เพียง 128 ตัวอักษรเท่านั้น จึงไม่สามารถรองรับภาษาอื่น ๆ ของโลกได้ เช่น ภาษาไทย ภาษาจีน หรือภาษาญี่ปุ่น
\end{rem}

\subsection{รหัสยูนิโคด (Unicode)}
\index{รหัสยูนิโคด}
\label{ssec:unicode-representation}
ระบบรหัสยูนิโคด (Unicode) กำหนดจุดรหัส (Code Point) ให้แก่อักขระและสัญลักษณ์จากภาษาและระบบเขียนต่าง ๆ \textbf{UTF-8} เป็นการเข้ารหัสแบบ \textbf{ความยาวไบต์ปรับเปลี่ยนได้ (Variable-Length Encoding)} โดยอักขระ ASCII ใช้ 1 ไบต์ และอักขระไทยทั่วไปใช้ 3 ไบต์ จำนวนไบต์ที่ใช้จึงขึ้นอยู่กับอักขระที่จัดเก็บ~\parencite{patterson2020organization}

\begin{definition}[การเข้ารหัสยูนิโคด]
\label{def:unicode-encodings}
\index{ยูนิโคด!การเข้ารหัส}
ระบบยูนิโคดมีรูปแบบการเข้ารหัสหลักดังนี้
\begin{enumerate}
    \item \textbf{UTF-8:} ใช้ขนาด 1 ถึง 4 ไบต์แบบปรับเปลี่ยนตามประเภทอักขระ (เป็นมาตรฐานหลักบนเว็บและอินเทอร์เน็ต)
    \item \textbf{UTF-16:} ใช้ขนาด 2 ถึง 4 ไบต์ (ใช้เป็นมาตรฐานภายในของ Windows, Java และ C\#)
    \item \textbf{UTF-32:} ใช้ขนาด 4 ไบต์คงที่ต่อหนึ่งอักขระ
\end{enumerate}
\end{definition}

\begin{ex}[การเข้ารหัสอักขระไทยในระบบ UTF-8]
\label{ex:utf8-thai}
จงแสดงการเข้ารหัสอักขระไทย `ก' (U+0E01) และ `ข' (U+0E02) ในระบบ UTF-8 แบบ 3 ไบต์
\end{ex}
\begin{solution}
\begin{align*}
\text{`ก' (U+0E01)} &= \texttt{0xE0 0xB8 0x81} \quad (\text{3 ไบต์})\\
\text{`ข' (U+0E02)} &= \texttt{0xE0 0xB8 0x82} \quad (\text{3 ไบต์})
\end{align*}
$\therefore$ คำว่า ``กข'' เข้ารหัสได้เป็น \texttt{0xE0B881E0B882} \solhere
\end{solution}

\sectionbanner{สรุปบทเรียน}
\label{sec:chapter-summary-numbase}
บทนี้เริ่มจากระบบเลขฐานสิบที่มนุษย์ใช้ในชีวิตประจำวัน แล้วเข้าสู่ระบบเลขฐานสองซึ่งเป็นภาษาของคอมพิวเตอร์ ฐานแปดและฐานสิบหกช่วยให้เขียนเลขฐานสองได้กระชับและแปลงกลับไปมากับฐานสองได้สะดวก การแปลงระหว่างฐานอาศัยการหารต่อเนื่องสำหรับส่วนจำนวนเต็มและการคูณต่อเนื่องสำหรับส่วนทศนิยม ถัดจากนั้นเป็นการบวก ลบ คูณ และหารในฐานสอง ก่อนปิดท้ายด้วยการแทนข้อมูลอักขระและสัญลักษณ์ด้วยรหัส ASCII และ Unicode

ความเข้าใจในระบบเลขฐานเป็นพื้นฐานสำหรับการศึกษาสถาปัตยกรรมคอมพิวเตอร์ การเขียนโปรแกรมระดับต่ำ และการเข้ารหัสข้อมูล~\parencite{rosen2019discrete}
