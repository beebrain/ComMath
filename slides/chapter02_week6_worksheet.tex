\noindent\textbf{กิจกรรมที่ 1 : เพาเวอร์เซตและความสัมพันธ์เชิงโครงสร้าง (Power Set Logic)} \hfill \textit{(6 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item กำหนด $A = \{1, \{2, 3\}\}$ จงแจกแจงสมาชิกทุกตัวของ $\mathcal{P}(A)$ ให้ครบถ้วน\\[0.15em]
        $\mathcal{P}(A) =$ \makebox[13.5cm]{\dotfill}
  \item กำหนด $B = \{a, b, c, d\}$ จงหา:
        \quad (ก) $|\mathcal{P}(B)| =$ \makebox[1.8cm]{\dotfill}
        \quad (ข) จำนวนเซตย่อยขนาด 2 สมาชิก $=$ \makebox[1.8cm]{\dotfill}
        \quad (ค) จำนวนเซตย่อยที่ \textbf{มี $a$ เป็นสมาชิก} $=$ \makebox[1.8cm]{\dotfill}
  \item ใส่เครื่องหมาย \textbf{T} (จริง) หรือ \textbf{F} (เท็จ):
        \quad [ \ ] $\emptyset \in \mathcal{P}(\mathcal{P}(\emptyset))$ \quad
        \quad [ \ ] $\{\emptyset\} \in \mathcal{P}(\mathcal{P}(\emptyset))$ \quad
        \quad [ \ ] $\{1\} \subseteq \mathcal{P}(\{1, 2\})$ \quad
        \quad [ \ ] $|\mathcal{P}(\mathcal{P}(\mathcal{P}(\emptyset)))| = 4$
\end{enumerate}
