%% ---------- กิจกรรม 2 ----------
\noindent\textbf{กิจกรรมที่ 2 : อนุกรมกับการวิเคราะห์ความซับซ้อนของขั้นตอนวิธี (Series \& Algorithm Complexity)} \hfill \textit{(8 นาที)}
\begin{enumerate}[label=\arabic*)]
  \item \textbf{การหาผลบวกวงวนซ้อน (Nested Loop Analysis):} คำนวณผลบวก $S = \sum_{i=1}^n i = 1 + 2 + 3 + \cdots + n$
        \[
        S_n = \frac{n(n+1)}{2} = \frac{1}{2}n^2 + \frac{1}{2}n \implies \text{Big-O Complexity} = O(\makebox[2cm]{\dotfill})
        \]
  \item หากขั้นตอนวิธี Bubble Sort ทำการเปรียบเทียบในรอบที่ 1 ถึง $n-1$ ทั้งหมด $S = (n-1) + (n-2) + \dots + 1$\\
        สำหรับข้อมูล $n = 100$ ตัว จะมีการเปรียบเทียบทั้งหมด $= \frac{(100)(99)}{2} =$ \makebox[3cm]{\dotfill} ครั้ง
  \item \textbf{ผลรวมจำนวนโหนดใน Binary Tree สมบูรณ์:} ต้นไม้ทวิภาคความสูง $h = 5$ มีโหนดในแต่ละชั้นคือ $1, 2, 4, 8, 16, 32$\\
        ผลรวมโหนดทั้งหมด $= S_6 = \frac{1(1 - 2^6)}{1 - 2} = 2^6 - 1 =$ \makebox[2.5cm]{\dotfill} โหนด
\end{enumerate}
\vspace{0.2em}

\end{enumerate}
